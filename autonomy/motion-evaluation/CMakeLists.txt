cmake_minimum_required(VERSION 3.16)
project(waymo_native_motion_metrics LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Protobuf REQUIRED)
find_package(absl REQUIRED)
find_package(glog REQUIRED)
add_subdirectory(/usr/src/googletest /motion-build/gtest)
file(GLOB_RECURSE PROTO_CC /generated/*.pb.cc)
add_library(wod_motion_proto ${PROTO_CC})
target_include_directories(wod_motion_proto PUBLIC /generated /upstream/src)
target_link_libraries(wod_motion_proto PUBLIC protobuf::libprotobuf)
set(WOD /upstream/src/waymo_open_dataset)
add_library(wod_motion_metrics
 ${WOD}/metrics/motion_metrics.cc ${WOD}/metrics/motion_metrics_utils.cc
 ${WOD}/common/status.cc ${WOD}/math/box2d.cc ${WOD}/math/polygon2d.cc ${WOD}/math/segment2d.cc)
target_link_libraries(wod_motion_metrics PUBLIC wod_motion_proto glog::glog absl::strings absl::str_format absl::flat_hash_map absl::status)
foreach(metric motion_metrics motion_metrics_utils)
 add_executable(${metric}_test ${WOD}/metrics/${metric}_test.cc)
 target_link_libraries(${metric}_test PRIVATE wod_motion_metrics GTest::gtest_main GTest::gmock)
endforeach()
